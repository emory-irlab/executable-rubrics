Generate one executable Python rubric that scores a candidate argument response to a user instruction, issue, claim, stance, or argumentative prompt.

Natural-language rubric:
Helpfulness — Assesses whether the response is useful for the user's request and meaningfully advances the task.

def score_helpfulness(query: str, text: str):
    max_possible = 20
    score = 0
    reasons = []

    q = (query or "").strip().lower()
    t = (text or "").strip()
    tl = t.lower()
    q_words = set(re.findall(r"[a-z0-9][a-z0-9_-]{2,}", q))
    t_words = set(re.findall(r"[a-z0-9][a-z0-9_-]{2,}", tl))
    overlap = len(q_words & t_words) / max(1, len(q_words))
    word_count = len(re.findall(r"\b\w+\b", t))

    refusal_markers = [
        "i can't", "i cannot", "i won’t", "i will not", "not able to", "unable to",
        "as an ai", "i do not have access", "i don't have access",
    ]
    task_action_markers = [
        "first", "second", "third", "steps", "example", "because", "therefore", "use",
        "try", "recommend", "solution", "answer", "here", "below", "summary", "code",
    ]
    asks_for_code = bool(re.search(r"\b(code|python|script|function|class|program|sql|regex|bash|javascript)\b", q))
    asks_for_list = bool(re.search(r"\b(list|examples|options|ideas|recommendations|compare|pros|cons)\b", q))
    asks_for_short = bool(re.search(r"\b(short|brief|concise|one sentence|one-line|tl;dr|quick)\b", q))

    if word_count >= 3:
        score += 3
        reasons.append("Response is non-empty and contains substantive content.")
    else:
        reasons.append("Response is empty or nearly empty.")

    if overlap >= 0.20 or not q_words:
        score += 4
        reasons.append("Response uses key concepts from the prompt and appears task-relevant.")
    elif overlap >= 0.08:
        score += 2
        reasons.append("Response has some topical overlap with the prompt.")
    else:
        reasons.append("Response has little lexical evidence of addressing the prompt.")

    if any(marker in tl for marker in task_action_markers) or asks_for_short:
        score += 4
        reasons.append("Response appears to move the task forward rather than only acknowledging it.")
    else:
        reasons.append("Response gives limited evidence of directly advancing the user's task.")

    if asks_for_code:
        has_code_signal = bool(re.search(r"\b(def|class|import|return|for |while |if |try:|except|SELECT|function)\b", t))
        if has_code_signal:
            score += 3
            reasons.append("Prompt appears to request code, and the response contains code-like content.")
        else:
            reasons.append("Prompt appears to request code, but the response lacks clear code-like content.")
    elif asks_for_list:
        has_list_signal = bool(re.search(r"(^|\n)\s*(-|\*|\d+[.)])\s+", t)) or ";" in t or "," in t
        if has_list_signal:
            score += 3
            reasons.append("Prompt appears to request multiple items, and the response provides a structured set of items.")
        else:
            reasons.append("Prompt appears to request multiple items, but the response is not clearly structured as such.")
    else:
        score += 3
        reasons.append("Response format is reasonably compatible with the prompt type.")

    if any(marker in tl for marker in refusal_markers):
        if re.search(r"\b(illegal|harm|unsafe|weapon|malware|fraud|privacy|medical emergency)\b", q):
            score += 2
            reasons.append("Response may appropriately refuse or limit unsafe/high-risk assistance.")
        else:
            score -= 3
            reasons.append("Response contains refusal/access-limit language that may be unhelpful for an ordinary task.")
    else:
        score += 3
        reasons.append("Response does not rely on unnecessary refusal or access-limit language.")

    filler_patterns = [
        r"^\s*(sure|okay|ok|got it)[.!\s]*$",
        r"i hope this helps",
        r"let me know if you (need|want)",
        r"as an ai language model",
    ]
    filler_hits = sum(1 for pattern in filler_patterns if re.search(pattern, tl))
    if filler_hits == 0:
        score += 3
        reasons.append("Response avoids obvious filler-only phrasing.")
    else:
        score -= min(3, filler_hits)
        reasons.append("Response includes filler or boilerplate that reduces usefulness.")

    score = max(0, min(max_possible, score))
    return {"score": score, "max_possible": max_possible, "reasons": reasons}


Natural-language rubric:
Coherence — Assesses whether the response is clear, logically organized, internally consistent, and easy to follow.

Executable rubric function:

def score_coherence(query: str, text: str):
    max_possible = 20
    score = 0
    reasons = []

    t = (text or "").strip()
    tl = t.lower()
    sentences = [s.strip() for s in re.split(r"[.!?]+|\n+", t) if s.strip()]
    words = re.findall(r"\b\w+\b", t)
    word_count = len(words)
    avg_sentence_len = word_count / max(1, len(sentences))

    if word_count >= 5:
        score += 3
        reasons.append("Response has enough words to form a coherent answer.")
    else:
        reasons.append("Response is too short to demonstrate coherence.")

    if 5 <= avg_sentence_len <= 35 or word_count < 20:
        score += 4
        reasons.append("Sentence length is generally readable.")
    elif avg_sentence_len <= 55:
        score += 2
        reasons.append("Sentence length is somewhat long but still potentially readable.")
    else:
        score -= 2
        reasons.append("Sentences are very long, which may hurt readability.")

    has_structure = bool(re.search(r"(^|\n)\s*(-|\*|\d+[.)])\s+\S+", t)) or len(sentences) >= 2 or ":" in t
    if has_structure:
        score += 4
        reasons.append("Response has visible structure or multiple coherent units.")
    else:
        reasons.append("Response has little visible structure.")

    transition_markers = [
        "because", "therefore", "however", "also", "first", "second", "finally", "for example",
        "in contrast", "so", "then", "instead", "while", "if", "when",
    ]
    if any(marker in tl for marker in transition_markers):
        score += 3
        reasons.append("Response uses connective language that supports logical flow.")
    elif word_count < 25:
        score += 2
        reasons.append("Short response can remain coherent without explicit transitions.")
    else:
        reasons.append("Response has limited connective language for its length.")

    malformed_signals = ["????", "!!!!", "... ...", "asdf", "lorem ipsum", "undefined undefined"]
    if not any(signal in tl for signal in malformed_signals):
        score += 3
        reasons.append("Response avoids obvious malformed or placeholder text.")
    else:
        score -= 3
        reasons.append("Response contains malformed or placeholder text.")

    repeated_lines = len(sentences) - len(set(s.lower() for s in sentences))
    if repeated_lines == 0:
        score += 3
        reasons.append("Response avoids repeated sentence-level content.")
    else:
        score -= min(3, repeated_lines)
        reasons.append("Response repeats sentence-level content, reducing coherence.")

    score = max(0, min(max_possible, score))
    return {"score": score, "max_possible": max_possible, "reasons": reasons}

def score_correctness(query: str, text: str):
    max_possible = 20
    score = 0
    reasons = []

    q = (query or "").strip().lower()
    t = (text or "").strip()
    tl = t.lower()
    word_count = len(re.findall(r"\b\w+\b", t))
    factual_prompt = bool(re.search(r"\b(what|who|when|where|why|how many|explain|define|is it true|fact|source|cite|compare|difference|best|latest|current)\b", q))
    code_prompt = bool(re.search(r"\b(code|python|script|function|class|program|sql|regex|bash|javascript|bug|error|traceback)\b", q))
    high_risk_prompt = bool(re.search(r"\b(medical|health|legal|visa|immigration|financial|investment|tax|safety|security)\b", q))

    uncertainty_markers = [
        "may", "might", "can", "could", "typically", "usually", "in general",
        "depends", "check", "verify", "not sure", "uncertain",
    ]
    overclaim_markers = [
        "always", "never", "guaranteed", "definitely", "certainly", "100%", "no doubt",
        "everyone knows", "impossible unless",
    ]
    unsupported_current_markers = ["latest", "currently", "today", "now", "as of", "recently", "this year"]

    if word_count >= 3:
        score += 3
        reasons.append("Response contains enough content to evaluate for correctness.")
    else:
        reasons.append("Response is too short to establish correctness.")

    if not re.search(r"\b([a-z]+)\b(?:\W+\1\b){3,}", tl):
        score += 3
        reasons.append("Response does not show obvious repetitive corruption.")
    else:
        score -= 3
        reasons.append("Response contains repeated wording that suggests low reliability.")

    if not any(marker in tl for marker in overclaim_markers):
        score += 4
        reasons.append("Response avoids strong absolute claims that often signal overstatement.")
    else:
        score -= 2
        reasons.append("Response uses absolute language that may overstate certainty.")

    if factual_prompt or high_risk_prompt:
        if any(marker in tl for marker in uncertainty_markers) or re.search(r"\b(source|evidence|according to|verify|consult|check)\b", tl):
            score += 4
            reasons.append("For a factual or high-stakes prompt, the response includes appropriate caution or verification language.")
        elif word_count < 30:
            score += 1
            reasons.append("For a factual or high-stakes prompt, the response is brief and gives little support for its claims.")
        else:
            score += 2
            reasons.append("For a factual or high-stakes prompt, the response gives some answer but limited visible grounding.")
    else:
        score += 4
        reasons.append("Prompt does not strongly require external factual grounding, so internal plausibility is the main correctness signal.")

    if code_prompt:
        balanced = t.count("(") == t.count(")") and t.count("[") == t.count("]") and t.count("{") == t.count("}")
        if balanced and not re.search(r"\b(TODO|pass\s*(#|$)|your code here|placeholder)\b", t):
            score += 4
            reasons.append("Code-oriented response has balanced delimiters and avoids obvious placeholders.")
        else:
            score -= 2
            reasons.append("Code-oriented response has likely syntax/placeholder issues.")
    else:
        score += 2
        reasons.append("No code-specific correctness constraints detected.")

    if any(marker in q for marker in unsupported_current_markers):
        if re.search(r"\b(as of|current|latest|verify|check|source|date)\b", tl):
            score += 2
            reasons.append("Prompt appears time-sensitive, and the response signals date/currentness awareness.")
        else:
            score -= 2
            reasons.append("Prompt appears time-sensitive, but the response does not signal date/currentness awareness.")
    else:
        score += 2
        reasons.append("Prompt does not appear strongly time-sensitive.")

    contradictions = [
        ("yes", "no"),
        ("true", "false"),
        ("increase", "decrease"),
        ("safe", "dangerous"),
        ("required", "optional"),
    ]
    contradiction_hits = sum(1 for a, b in contradictions if re.search(r"\b" + a + r"\b", tl) and re.search(r"\b" + b + r"\b", tl))
    if contradiction_hits == 0:
        score += 2
        reasons.append("No simple internal contradiction markers were detected.")
    else:
        score -= min(4, 2 * contradiction_hits)
        reasons.append("Response contains simple opposing claims that may indicate inconsistency.")

    score = max(0, min(max_possible, score))
    return {"score": score, "max_possible": max_possible, "reasons": reasons}
