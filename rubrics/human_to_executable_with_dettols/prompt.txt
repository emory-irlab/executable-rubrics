        reasons.append("Response is nearly empty, with very limited useful content.")

            reasons.append("Response is substantive but too verbose for a concise request.")

        elif has_direct_answer:

            reasons.append("Response somewhat answers the yes/no-style request.")
        else:
            reasons.append("Response does not directly answer the yes/no-style request.")
    elif asks_to_rewrite:

            reasons.append("Response provides rewritten or revised content rather than only discussing the task.")

        score += 2
        reasons.append("Response gives a concrete answer rather than only meta-commentary.")

        score += 1
        reasons.append("Response contains useful content but could be more direct.")

    if response_refuses_or_redirects(text) and not refusal_matches_policy_need(query, text):

        reasons.append("Response gives actionable, structured help that advances the task.")

        score += 2
        reasons.append("Response gives some actionable help or useful structure.")

        reasons.append("Response provides relevant information, but limited direct actionability.")

        constraint_checks += 1
        needed = int(min_items_match.group(1))

            reasons.append("Response satisfies the requested minimum number of items.")
        else:
            reasons.append("Response does not provide the requested minimum number of items.")

        constraint_checks += 1
        needed = int(exact_items_match.group(1))
        if item_count == needed:

            reasons.append("Response does not provide exactly the requested number of items.")

        if all(normalize_text(term) in t_norm or token_set_ratio(term, text) >= 75 for term in include_terms):

        if has_code_block and has_balanced_fences and (has_requested_language or looks_like_requested):
            format_score = 3
            reasons.append("Prompt requests code, and the response provides well-formed code in an appropriate format.")
        elif has_code_block or looks_like_requested:
            format_score = 2
            reasons.append("Prompt requests code, and the response contains code-like content.")
        else:
            reasons.append("Prompt requests code, but the response lacks clear usable code.")

    elif asks_for_table:
        if contains_markdown_table(text):
            format_score = 3
            reasons.append("Prompt calls for comparison or tabular organization, and the response provides a table.")
        elif count_numbered_items(text) >= 2:
            format_score = 2
            reasons.append("Prompt calls for organized comparison, and the response provides structured items.")
        else:
            format_score = 0.75
            reasons.append("Prompt would benefit from structured comparison, but the response is weakly organized.")

    elif asks_for_list:
        item_count = count_numbered_items(text)
        if item_count >= 3:
            format_score = 3
            reasons.append("Prompt requests multiple items, and the response provides a clear structured list.")
        elif item_count >= 2 or ";" in text:
            format_score = 2
            reasons.append("Prompt requests multiple items, and the response provides some structured options.")
        else:
            format_score = 0.75
            reasons.append("Prompt requests multiple items, but the response is not clearly structured.")

    elif asks_for_calculation:
        if has_stepwise_calculation(text):
            format_score = 3
            reasons.append("Prompt appears to require calculation or reasoning, and the response shows stepwise work.")
        elif re.search(r"\d", text):
            format_score = 1.5
            reasons.append("Prompt appears calculation-oriented, and the response includes numeric content.")
        else:
            format_score = 0.75
            reasons.append("Prompt appears calculation-oriented, but the response lacks visible computation.")

    elif asks_for_steps:
        if count_numbered_items(text) >= 2:
            format_score = 3
            reasons.append("Prompt asks for guidance, and the response gives clear steps.")
        elif has_action_language:
            format_score = 2
            reasons.append("Prompt asks for guidance, and the response gives actionable advice.")
        else:
            format_score = 0.75
            reasons.append("Prompt asks for guidance, but the response is not very actionable.")

    else:
        format_score = 2.25
        reasons.append("Response format is reasonably compatible with the prompt type.")

    score += format_score

    specificity_score = 0.0

    has_specifics = (
        bool(re.search(r"\b\d+(\.\d+)?\b", text))
        or count_numbered_items(text) >= 2
        or contains_markdown_table(text)
        or count_markdown_code_blocks(text) >= 1
        or bool(re.search(r"\b(for example|e\.g\.|because|so that|instead of|specifically)\b", t_norm))
    )

    generic_phrases = [
        "it depends",
        "there are many ways",
        "you should consider",
        "do some research",
        "look it up",
        "use common sense",
        "various factors",
    ]
    generic_count = sum(1 for phrase in generic_phrases if phrase in t_norm)

    if asks_for_short and has_direct_answer:
        specificity_score += 2
        reasons.append("For a concise request, the response gives enough direct detail.")
    elif has_specifics and generic_count == 0:
        specificity_score += 2
        reasons.append("Response includes concrete details that make it more practically useful.")
    elif has_specifics:
        specificity_score += 1.25
        reasons.append("Response includes some concrete detail, though it may still be partly generic.")
    elif word_count >= 40 and generic_count <= 1:
        specificity_score += 1
        reasons.append("Response has some explanatory substance, but limited concrete specificity.")
    else:
        reasons.append("Response is too generic to be maximally helpful.")

    score += specificity_score

    # 8) Refusal, safety, and caveat handling: 2 points
    refuses = response_refuses_or_redirects(text)
    appropriate_refusal = refusal_matches_policy_need(query, text)
    gives_actionable_harm = response_provides_actionable_harm(text)

    unnecessary_caveats = [
        "i am not a lawyer",
        "i am not a doctor",
        "i am not a financial advisor",
        "as an ai",
        "i don't have access",
        "i do not have access",
        "i can't browse",
        "i cannot browse",
    ]
    caveat_count = sum(1 for phrase in unnecessary_caveats if phrase in t_norm)

    if appropriate_refusal:
        score += 2
        reasons.append("Response appropriately refuses or redirects a harmful request without actionable harm.")
    elif gives_actionable_harm:
        score -= 4
        reasons.append("Response appears to provide actionable harmful content, which severely reduces helpfulness.")
    elif refuses:
        score -= 2
        reasons.append("Response contains refusal or access-limit language that appears unnecessary for this request.")
    elif caveat_count > 0 and not re.search(r"\b(medical|legal|financial|health|emergency|diagnosis|contract|tax)\b", q_norm):
        score -= min(1.5, 0.5 * caveat_count)
        reasons.append("Response includes unnecessary caveats that reduce usefulness.")
    else:
        score += 2
        reasons.append("Response avoids unnecessary refusal and excessive caveats.")

    # 9) Clarity, language match, and boilerplate: 1.5 points
    quality_points = 0.0

    filler_patterns = [
        r"^\s*(sure|okay|ok|got it)[.!\s]*$",
        r"i hope this helps",
        r"let me know if you (need|want)",
        r"as an ai language model",
        r"i am an ai language model",
    ]
    filler_hits = sum(1 for pattern in filler_patterns if re.search(pattern, t_norm))

    if language_matches_query(query, text):
        quality_points += 0.5
        reasons.append("Response language appears to match the user's language.")
    else:
        reasons.append("Response language may not match the user's language.")

    if filler_hits == 0:
        quality_points += 0.5
        reasons.append("Response avoids obvious filler and boilerplate.")
    else:
        quality_points -= min(1.0, 0.5 * filler_hits)
        reasons.append("Response includes filler or boilerplate that reduces usefulness.")

    avg_words_per_sentence = word_count / sentence_ct
    if 4 <= avg_words_per_sentence <= 35:
        quality_points += 0.5
        reasons.append("Response is reasonably readable.")
    else:
        reasons.append("Response may be awkwardly terse or overly dense.")

    if re.search(r"\b(help|advice|recommend|explain|fix|problem|issue|worried|confused)\b", q_norm):
        if is_supportive_tone(text):
            quality_points += 0.25
            reasons.append("Response tone appears reasonably supportive.")
        else:
            quality_points -= 0.25
            reasons.append("Response tone may be insufficiently supportive for the request.")

    score += max(0.0, min(1.5, quality_points))

    score = int(round(max(0, min(max_possible, score))))
    return {
        "score": score,
        "max_possible": max_possible,
        "reasons": reasons,
    }

Coherence — Assesses whether the response is clear, logically organized, internally consistent, and easy to follow.

Executable rubric function:
import re
import math
from itertools import combinations

def score_coherence(query: str, text: str):
    """
    Scores whether a response is clear, logically organized, internally consistent,
    and easy to follow.

    Assumes the helper functions from the provided tool set are available:
      normalize_text, sentence_count, count_numbered_items, contains_markdown_table,
      count_markdown_code_blocks, has_balanced_markdown_fences,
      extract_markdown_code_blocks, token_set_ratio, sequence_similarity,
      contains_pattern, count_pattern_matches, grammar_errors,
      flesch_reading_ease, flesch_kincaid_grade.
    """
    max_possible = 20
    score = 0.0
    reasons = []

    query = query or ""
    text = text or ""

    t = text.strip()
    t_norm = normalize_text(t)
    q_norm = normalize_text(query)

    words = re.findall(r"\b\w+\b", t)
    word_count = len(words)

    # Sentence splitting with fallback to tool-provided sentence_count.
    rough_sentences = [
        s.strip()
        for s in re.split(r"(?<=[.!?])\s+|\n+", t)
        if s.strip()
    ]
    sent_count = max(
        1,
        sentence_count(t) or len(rough_sentences) or 1
    )
    avg_sentence_len = word_count / sent_count

    paragraphs = [p.strip() for p in re.split(r"\n\s*\n", t) if p.strip()]
    line_items = count_numbered_items(t)
    has_table = contains_markdown_table(t)
    code_block_count = count_markdown_code_blocks(t)
    has_code = code_block_count > 0
    balanced_fences = has_balanced_markdown_fences(t)

    asks_for_code = bool(re.search(
        r"\b(code|python|script|function|class|program|sql|regex|bash|javascript|html|typescript|rust)\b",
        q_norm,
    ))
    asks_for_steps_or_structure = bool(re.search(
        r"\b(steps|list|outline|plan|compare|pros|cons|table|explain|guide|walkthrough|organize)\b",
        q_norm,
    ))
    asks_for_short = bool(re.search(
        r"\b(short|brief|concise|one sentence|one-line|tl;?dr|quick|succinct)\b",
        q_norm,
    ))

    # 1) Minimum substance: 2 points
    if word_count == 0:
        reasons.append("Response is empty.")
    elif word_count < 4:
        score += 0.5
        reasons.append("Response is very short, so coherence is hard to establish.")
    elif asks_for_short and word_count <= 80:
        score += 2
        reasons.append("Response has enough substance for a concise answer.")
    elif word_count >= 8:
        score += 2
        reasons.append("Response has enough content to demonstrate coherence.")
    else:
        score += 1
        reasons.append("Response has limited content but is not empty.")

    # 2) Readability and sentence length: 3 points
    readability_points = 0.0

    if word_count < 20:
        readability_points += 2
        reasons.append("Short response can remain readable without complex sentence structure.")
    elif 5 <= avg_sentence_len <= 30:
        readability_points += 2
        reasons.append("Average sentence length is easy to follow.")
    elif 30 < avg_sentence_len <= 45:
        readability_points += 1
        reasons.append("Average sentence length is somewhat long but still readable.")
    else:
        readability_points -= 1
        reasons.append("Sentence length may make the response hard to follow.")

    try:
        fk_grade = flesch_kincaid_grade(t)
        reading_ease = flesch_reading_ease(t)

        # These helpers may return 0.0 when unavailable, so treat 0 as neutral.
        if reading_ease:
            if reading_ease >= 45:
                readability_points += 1
                reasons.append("Readability score suggests the response is reasonably accessible.")
            elif reading_ease >= 25:
                readability_points += 0.5
                reasons.append("Readability score suggests moderate difficulty.")
            else:
                reasons.append("Readability score suggests the response may be difficult to read.")
        elif fk_grade:
            if fk_grade <= 12:
                readability_points += 1
                reasons.append("Grade-level estimate suggests the response is reasonably readable.")
            elif fk_grade <= 16:
                readability_points += 0.5
                reasons.append("Grade-level estimate suggests moderate complexity.")
            else:
                reasons.append("Grade-level estimate suggests high complexity.")

The generated executable rubric must satisfy the following interface:

def score_response(query: str, text: str):
    ...

It must return a dictionary with exactly these keys:
- "score"
- "max_possible"
- "normalized_score"
- "reasons"

Each of the fourteen axis-specific functions should return a dictionary with:
- "score"
- "max_possible"
- "reasons"

The top-level score_response function should:
1. Call all fourteen axis-specific rubric functions.
2. Sum their "score" values.
3. Sum their "max_possible" values.
4. Compute normalized_score as score / max_possible.
5. Concatenate or group the reasons from all fourteen axis functions.
6. Return the final dictionary.

Requirements:
1. The same rubric must apply to any ArgQuality query-argument pair.
2. Use only deterministic Python logic. No file I/O, no network calls, no external APIs, and no LLM calls.
3. Do not rely on intermediate conversations. Only use query and text.
4. Reward clear claims, relevant premises, acceptable premises, sufficient support, credible wording, appropriate emotional appeal, coherent arrangement, reasonable framing, issue relevance, and counterargument handling.
5. Penalize unsupported assertions, irrelevant premises, false or implausible claims, weak evidence, circular reasoning, strawman arguments, ad hominem attacks, excessive emotional manipulation, unclear wording, disorganized structure, off-topic content, and failure to address the issue or stance.
6. Use partial credit within each axis.
7. max_possible must be a single numeric literal in score_response equal to the sum of positive points from the fourteen axis functions.
8. Each axis-specific max_possible must also be a numeric literal.
9. normalized_score must be score / max_possible.
10. The rubric should be general enough to transfer across the ArgQuality corpus while still being specific enough to separate better and worse arguments.
11. Encode important dependencies with explicit if/else logic. For example:
    - If the query contains an issue or stance, reward arguments that clearly address that issue or stance.
    - If the response makes a conclusion, reward premises that are relevant to that conclusion.
    - If the response contains several premises, reward diversity and sufficiency of support rather than repetition.
    - If the argument makes strong factual claims, reward cautious wording, evidence markers, examples, statistics, or source-like support.
    - If the issue is controversial, reward acknowledgement of tradeoffs and anticipated counterarguments.
    - If the response uses emotional language, reward proportional and audience-appropriate emotion but penalize fearmongering, insults, or manipulation.
    - If the argument has no identifiable conclusion, penalize cogency, arrangement, local relevance, and global relevance.
    - If the argument is mostly off-topic, strongly penalize global relevance, local relevance, effectiveness, and reasonableness.
12. Prefer deterministic helper tools whenever they fit the need instead of rewriting matching logic from scratch.
13. Feel free to use regexes, dictionaries, or other Python objects necessary to expand coverage. Ensure that when you define strings as part of lists, sets, dictionaries, etc., they do not only match exact variants.
14. Be careful not to overmatch, as that might wrongly assign similar scores to adversarial or bad arguments.
15. Clamp each axis score between 0 and that axis's max_possible.
16. Clamp the final score between 0 and the final max_possible.

def _fallback_quantities(text: str):
    regex = _require("regex", "regex")
    if isinstance(regex, _MockModule):
        import re as regex
    pattern = regex.compile(
        r"(?P<surface>(?P<value>[+-]?\d+(?:\.\d+)?)\s*(?P<unit>%|[A-Za-zμ]+)?)",
        flags=regex.IGNORECASE,
    )
    quantities = []
    for match in pattern.finditer(text or ""):
        surface = (match.group("surface") or "").strip()
        value_text = match.group("value")
        if not surface or not value_text:
            continue
        try:
            value = float(value_text)
        except ValueError:
            continue
        unit_text = (match.group("unit") or "").strip()
        normalized_unit = _FALLBACK_UNIT_ALIASES.get(unit_text.lower()) if unit_text else None
        unit_obj = SimpleNamespace(name=normalized_unit) if normalized_unit else None
        quantities.append(SimpleNamespace(surface=surface, value=value, unit=unit_obj))
    return quantities


def normalize_text(text: str) -> str:
    regex = _require("regex", "regex")
    if isinstance(regex, _MockModule):
        import re as regex
    text = unicodedata.normalize("NFKD", text or "")
    text = "".join(ch for ch in text if not unicodedata.combining(ch))
    text = text.lower()
    if hasattr(regex, 'sub'):
        # Fallback to standard re if regex module is missing or limited
        try:
            text = regex.sub(r"[^\w\d%./+\-\s]", " ", text)
        except:
             import re
             text = re.sub(r"[^\w\d%./+\-\s]", " ", text)
    text = regex.sub(r"\s+", " ", text).strip()
    return text


def strip_markdown_fences(text: str) -> str:

        import re as regex
    text = text or ""
    text = regex.sub(r"^```[\w+-]*\s*", "", text.strip())
    text = regex.sub(r"\s*```$", "", text.strip())
    return text.strip()


def extract_markdown_code_blocks(text: str) -> list[dict[str, str]]:

def extract_role_messages(conversation: str) -> list[tuple[str, str]]:
    messages: list[tuple[str, str]] = []
    current_role = None
    current_lines: list[str] = []
    for raw_line in (conversation or "").splitlines():
        line = raw_line.rstrip()
        lower = line.lower()
        if lower.startswith(("user:", "assistant:", "developer:", "system:")):
            if current_role is not None:
                messages.append((current_role, "\n".join(current_lines).strip()))
            role, content = line.split(":", 1)
            current_role = role.strip().lower()
            current_lines = [content.strip()]
        else:
            if current_role is not None:
                current_lines.append(line)
    if current_role is not None:
        messages.append((current_role, "\n".join(current_lines).strip()))
    return messages


def extract_last_role_message(conversation: str, role: str) -> str:
    wanted = normalize_text(role)
    for msg_role, content in reversed(extract_role_messages(conversation)):
        if normalize_text(msg_role) == wanted:
            return content
    return ""

def jaccard_similarity(tokens_a: Iterable[str], tokens_b: Iterable[str]) -> float:
    textdistance = _require("textdistance", "textdistance")
    if isinstance(textdistance, _MockModule):
        s1 = set(tokens_a)
        s2 = set(tokens_b)
        return len(s1 & s2) / len(s1 | s2) if (s1 | s2) else 1.0
    return float(textdistance.jaccard.normalized_similarity(list(tokens_a), list(tokens_b)))

def exact_section_changed(candidate: str, reference: str, section_name: str) -> bool:
    def _extract(text: str) -> str:
        current = []
        in_section = False
        for line in (text or "").splitlines():
            if line.strip().lower().startswith(section_name.lower()):
                in_section = True
                continue
            if in_section and line.strip().endswith(":"):
                break
            if in_section:
                current.append(line)
        return "\n".join(current).strip()

    return _extract(candidate) != _extract(reference)

def contains_required_keyword(text: str, keywords: list[str]) -> bool:
    kp = _keyword_processor(keywords)
    if not kp:
        # Fallback to simple containment if flashtext is missing
        text_norm = normalize_text(text)
        return any(normalize_text(k) in text_norm for k in keywords)
    return bool(extract_keywords(text, kp))


def count_keyword_hits(text: str, keywords: list[str]) -> int:
    kp = _keyword_processor(keywords)
    if not kp:
        text_norm = normalize_text(text)
        return len({k for k in keywords if normalize_text(k) in text_norm})
    return len(set(extract_keywords(text, kp)))


def replace_keywords(text: str, mapping: dict[str, str]) -> str:
    processor = _keyword_processor(mapping.keys())
    if not processor: return text
    for key, value in mapping.items():
        processor.add_keyword(normalize_text(key), value)
    return processor.replace_keywords(normalize_text(text))


def keyword_coverage(text: str, required_keywords: list[str]) -> float:
    if not required_keywords:
        return 0.0
    return count_keyword_hits(text, required_keywords) / len({normalize_text(k) for k in required_keywords if normalize_text(k)})

def match_required_terms(text: str, required_terms: Iterable[str], allow_substrings: bool = False) -> bool:
    norm_text = normalize_text(text)
    norm_terms = [normalize_text(term) for term in required_terms if normalize_text(term)]
    if allow_substrings:
        return all(term in norm_text for term in norm_terms)
    return all(contains_required_keyword(text, [term]) for term in norm_terms)


def build_automaton(phrases: Iterable[str]):
    ahocorasick = _require("ahocorasick", "pyahocorasick")
    if isinstance(ahocorasick, _MockModule): return None
    automaton = ahocorasick.Automaton()
    for idx, phrase in enumerate(phrases):
        norm = normalize_text(phrase)
        if norm:
            automaton.add_word(norm, (idx, norm))
    automaton.make_automaton()
    return automaton

    doc = _negex_doc(text)
    if not doc: return []
    wanted = {normalize_text(concept) for concept in concepts}
    found = []
    for ent in doc.ents:

        if norm in wanted and (not hasattr(ent._, 'negex') or not ent._.negex):
            found.append(ent.text)
    return found

def grammar_errors(text: str):

def word_frequency_score(word: str, lang: str = "en") -> float:
    wordfreq = _require("wordfreq", "wordfreq")
    if isinstance(wordfreq, _MockModule): return 0.0
    return float(wordfreq.zipf_frequency(word or "", lang))


def average_word_frequency(text: str, lang: str = "en") -> float:
    tokens = [token for token in _tokenize(text) if token.strip()]
    if not tokens:
        return 0.0
    freqs = [word_frequency_score(token, lang) for token in tokens]
    return statistics.mean(freqs) if freqs else 0.0


def rare_word_count(text: str, lang: str = "en", threshold: float = 3.0) -> int:
    return sum(1 for token in _tokenize(text) if word_frequency_score(token, lang) < threshold)


def contains_jargon(text: str, jargon_terms: Iterable[str]) -> bool:
    jargon = {normalize_text(term) for term in jargon_terms}
    return any(normalize_text(token) in jargon for token in _tokenize(text))


def simplify_candidates(text: str, lang: str = "en", threshold: float = 3.0) -> list[str]:
    return [token for token in _tokenize(text) if word_frequency_score(token, lang) < threshold]


Benchmark-specific emphasis:
- Prioritize argument quality rather than factual domain expertise.
- Reward arguments that contain a clear conclusion, identifiable premises, and a logical relationship between premises and conclusion.
- Reward premises that are plausible, relevant, non-circular, and sufficient to support the stance.
- Reward arguments that address the issue in the query rather than switching topics.
- Reward credible, measured, and audience-aware style.
- Reward emotionally engaging language only when it supports the argument without replacing evidence.
- Reward coherent structure: issue framing, premises, support, counterargument or limitation where useful, and conclusion.
- Reward reasonable handling of opposing views, tradeoffs, exceptions, or anticipated objections.
- Penalize insults, profanity, exaggeration, unsupported absolutes, purely emotional appeals, vague generalities, repetition, and missing conclusions.
- Penalize arguments that merely restate the stance without giving reasons.
